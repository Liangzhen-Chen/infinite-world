// 必跑测试文件缺失（登记的写函数在，豁免用得上）：计（裁定②）
// @file tests/accept/temp-root.mjs
import { writeFileSync } from 'node:fs';
export function assertRelUnderTempRoot(root, rel) { return root + rel; }
export function writeUnderTempRoot(root, rel, data) {
  const full = assertRelUnderTempRoot(root, rel);
  writeFileSync(full, data);
}
