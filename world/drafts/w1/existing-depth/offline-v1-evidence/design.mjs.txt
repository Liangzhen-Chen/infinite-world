export const patchRows=base.map(before=>{
 const key=before.id.slice('npc_hw_'.length),row=authoredRows.find(r=>r[0]===key);
 if(!row)throw Error(`未起草 ${before.id}`);
 const issues=scopeAudit.issues.filter(i=>i.id===before.id),patches=[];
 const add=(path,value)=>patches.push({path,value});
 const events=sharedEvents.filter(e=>e.people.includes(key));
 if(events.length!==1)throw Error(`共享事件须唯一 ${before.id}`);
 for(const issue of issues){
  const {field,code}=issue;
  if(code==='known-count'){
   if(before.knowledge.known.length!==1)throw Error('冻结知识条数变化');
   const oldFact=key==='baihui'?'白蕙在陆青的小学堂学习识字，课外承接已经商定的田间活。':before.memorySeeds[1].event;
   add('knowledge.known[1]',{fact:oldFact,source:before.id,reliability:'亲历，原档已记经历'});
   add('knowledge.known[2]',{fact:events[0].event,source:before.id,reliability:'亲历'});
  }else if(code==='known-other')continue;
  else if(code==='unknown-specific')add(field,row[2][Number(field.match(/\d+/)[0])]);
  else if(code==='goal-steps')add(field,row[3][['short','mid','long'].indexOf(field.split('.')[1].split('[')[0])]);
  else if(code==='voice-register')add(field,row[1]);
  else if(code==='relationship-note'){
   const rel=before.relationships[Number(field.match(/\d+/)[0])];
   const text=relationshipNotes[key]?.[rel.with.slice('npc_hw_'.length)];
   if(!text)throw Error(`缺关系来源 ${before.id}/${field}`);add(field,text);
  }else if(code==='memory-shared'&&field==='memorySeeds'){
   const e=events[0];add(`memorySeeds[${before.memorySeeds.length}]`,{id:e.id,event:e.event,knownBy:e.knownBy,...(e.location?{location:e.location}:{}),salience:3,reliability:'亲历'});
  }else throw Error(`未处理失败叶 ${before.id}/${field}/${code}`);
 }
 const eventSources=events[0].sources;
 const referenceRelationshipIds=[...new Set([...eventSources.filter(s=>s[1].startsWith('relationships')).map(s=>id(s[0])),...(issues.some(i=>i.code==='relationship-note')?[before.id]:[])])];
 const referenceMemoryIds=[...new Set([...eventSources.filter(s=>s[1].startsWith('memorySeeds')).map(s=>id(s[0])),...(key==='baihui'?[]:[before.id])])];
 return{id:before.id,before,baseHash:hashContent(before),issues,patches,referenceRelationshipIds,referenceMemoryIds};
});
